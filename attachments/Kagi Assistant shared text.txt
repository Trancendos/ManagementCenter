4.2 Schema Template: ledger_service (Postgres-XL)
sql

-- ledger_service.sql (partitioned by data_class)
CREATE TABLE ledger_entries (
    id UUID PRIMARY KEY,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    data_class ENUM('internal', 'external') NOT NULL,  -- #474 partitioning
    payload JSONB NOT NULL,
    source_worker TEXT NOT NULL,
    checksum TEXT NOT NULL  -- SHA-256 of payload
) PARTITION BY LIST (data_class);

-- Internal partition (sensitive)
CREATE TABLE ledger_entries_internal PARTITION OF ledger_entries
    FOR VALUES IN ('internal');

-- External partition (auditable)
CREATE TABLE ledger_entries_external PARTITION OF ledger_entries
    FOR VALUES IN ('external');
